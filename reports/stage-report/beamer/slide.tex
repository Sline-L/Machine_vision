    \end{column}

\begin{frame}{当前视觉处理流程}
  \begin{enumerate}
    \item Model1 在整帧中定位齿轮并生成 ROI
    \item Scratch V5 与 Missing Hole V1 分别输出专项概率
    \item 两个专项独立过阈值，最终执行 OR 判定
    \item 同一结果进入 Web、统计和串口输出
  \end{enumerate}

  \vfill
  \begin{exampleblock}{阶段结论}
    软件闭环已经运行，真实设备条件下的端到端指标仍需验收。
  \end{exampleblock}
\end{frame}
